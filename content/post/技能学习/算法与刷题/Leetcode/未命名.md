```cpp
#include <bits/stdc++.h>
using namespace std;

struct Process {
    char type;                  // 'X', 'A', 'B', 'C'
    int st, w, k;               // 开始段、忍耐限度（X类为0）、任务数
    vector<pair<int,int>> tasks; // (申请资源, 持续段数)
    int state;                  // 0=未开始, 1=资源申请, 2=任务运行, 3=已结束
    int j;                      // 当前任务索引（0-based）
    int rem;                    // 当前任务剩余运行段数
    int fail_count;             // 连续申请失败次数
    int first_apply_seg;        // 首次申请当前资源的段号，-1表示未设置
    uint64_t owned;             // 位掩码，bit x 表示占有资源 x（1-based）
    long long profit;           // 总收益
    int start_seg;              // 开始活动段
    int end_seg;                // 结束活动段，-1表示未结束
    bool need_release_A;        // A 类标记：段末释放所有资源
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<Process> p(n);
    for (int i = 0; i < n; ++i) {
        char type;
        cin >> type;
        p[i].type = type;
        if (type == 'X') {
            cin >> p[i].st >> p[i].k;
            p[i].w = 0;
        } else {
            cin >> p[i].st >> p[i].w >> p[i].k;
        }
        p[i].tasks.resize(p[i].k);
        for (int j = 0; j < p[i].k; ++j) {
            cin >> p[i].tasks[j].first >> p[i].tasks[j].second;
        }
        // 初始化
        p[i].state = 0;
        p[i].j = 0;
        p[i].rem = 0;
        p[i].fail_count = 0;
        p[i].first_apply_seg = -1;
        p[i].owned = 0;
        p[i].profit = 0;
        p[i].start_seg = -1;
        p[i].end_seg = -1;
        p[i].need_release_A = false;
    }

    int owner[45] = {0};        // 资源占有者：1..n，0 表示待使用
    int seg = 1;
    const int MAX_SEG = 100000; // 足够大，实际不会超过几千段

    while (seg <= MAX_SEG) {
        // 检查是否所有进程都已结束
        bool all_done = true;
        for (int i = 0; i < n; ++i) {
            if (p[i].state != 3) { all_done = false; break; }
        }
        if (all_done) break;

        // ---------- 段首：开始活动与状态初始化 ----------
        for (int i = 0; i < n; ++i) {
            if (p[i].state == 0 && seg == p[i].st) {
                p[i].state = 1;
                p[i].j = 0;
                p[i].rem = 0;
                p[i].fail_count = 0;
                p[i].first_apply_seg = seg;
                p[i].profit = 0;
                p[i].start_seg = seg;
                p[i].end_seg = -1;
                p[i].owned = 0;
                p[i].need_release_A = false;
            }
            // 刚进入资源申请状态的进程，设置首次申请段
            if (p[i].state == 1 && p[i].first_apply_seg == -1) {
                p[i].first_apply_seg = seg;
            }
        }

        // ---------- 段首：收集各资源的申请 ----------
        vector<vector<int>> normal_apply(m + 1); // 普通申请
        vector<vector<int>> c_grab(m + 1);       // C 类抢夺
        for (int i = 0; i < n; ++i) {
            if (p[i].state == 1) {
                int res = p[i].tasks[p[i].j].first;
                if (p[i].type == 'C') {
                    // 判断是否满足 C 类抢夺条件
                    if (p[i].fail_count == p[i].w && seg == p[i].first_apply_seg + p[i].w) {
                        c_grab[res].push_back(i);
                    } else {
                        normal_apply[res].push_back(i);
                    }
                } else {
                    normal_apply[res].push_back(i);
                }
            }
        }

        // ---------- 段首：处理每个资源的申请与抢夺 ----------
        for (int res = 1; res <= m; ++res) {
            if (!c_grab[res].empty()) {
                // 有 C 类抢夺，忽略所有普通申请
                // 选择编号最大的 C 类进程
                int winner = -1;
                for (int idx : c_grab[res]) {
                    if (winner == -1 || idx > winner) winner = idx;
                }
                // 处理抢夺成功者
                {
                    int i = winner;
                    // 剥夺原占有者
                    if (owner[res] != 0) {
                        int old = owner[res] - 1;
                        p[old].owned &= ~(1ULL << res);
                    }
                    owner[res] = i + 1;
                    p[i].owned |= (1ULL << res);
                    p[i].state = 2;
                    p[i].rem = p[i].tasks[p[i].j].second;
                    p[i].fail_count = 0;
                    p[i].first_apply_seg = -1;
                }
                // 处理抢夺失败的 C 类进程
                for (int idx : c_grab[res]) {
                    if (idx == winner) continue;
                    int i = idx;
                    p[i].state = 2;
                    p[i].rem = p[i].tasks[p[i].j].second;
                    p[i].fail_count = 0;
                    p[i].first_apply_seg = -1;
                }
                // 所有普通申请均失败
                for (int idx : normal_apply[res]) {
                    int i = idx;
                    p[i].fail_count++;
                    if (p[i].fail_count == p[i].w) {
                        if (p[i].type == 'B') {
                            p[i].state = 2;
                            p[i].rem = p[i].tasks[p[i].j].second;
                            p[i].fail_count = 0;
                            p[i].first_apply_seg = -1;
                        } else if (p[i].type == 'A') {
                            p[i].need_release_A = true;
                        } else if (p[i].type == 'C') {
                            // 等待下一段抢夺
                        }
                    }
                }
            } else {
                // 无 C 类抢夺，处理普通申请
                if (!normal_apply[res].empty()) {
                    if (owner[res] == 0) {
                        // 资源待使用，选择编号最小的申请者
                        int winner = -1;
                        for (int idx : normal_apply[res]) {
                            if (winner == -1 || idx < winner) winner = idx;
                        }
                        int i = winner;
                        owner[res] = i + 1;
                        p[i].owned |= (1ULL << res);
                        p[i].state = 2;
                        p[i].rem = p[i].tasks[p[i].j].second;
                        p[i].fail_count = 0;
                        p[i].first_apply_seg = -1;
                        // 其他申请者失败
                        for (int idx : normal_apply[res]) {
                            if (idx == winner) continue;
                            int j = idx;
                            p[j].fail_count++;
                            if (p[j].fail_count == p[j].w) {
                                if (p[j].type == 'B') {
                                    p[j].state = 2;
                                    p[j].rem = p[j].tasks[p[j].j].second;
                                    p[j].fail_count = 0;
                                    p[j].first_apply_seg = -1;
                                } else if (p[j].type == 'A') {
                                    p[j].need_release_A = true;
                                } else if (p[j].type == 'C') {
                                    // 等待下一段抢夺
                                }
                            }
                        }
                    } else {
                        // 资源被占有，所有申请失败
                        for (int idx : normal_apply[res]) {
                            int i = idx;
                            p[i].fail_count++;
                            if (p[i].fail_count == p[i].w) {
                                if (p[i].type == 'B') {
                                    p[i].state = 2;
                                    p[i].rem = p[i].tasks[p[i].j].second;
                                    p[i].fail_count = 0;
                                    p[i].first_apply_seg = -1;
                                } else if (p[i].type == 'A') {
                                    p[i].need_release_A = true;
                                } else if (p[i].type == 'C') {
                                    // 等待下一段抢夺
                                }
                            }
                        }
                    }
                }
            }
        }

        // ---------- 段中：计算收益 ----------
        for (int i = 0; i < n; ++i) {
            if (p[i].state == 2) {
                p[i].profit += __builtin_popcountll(p[i].owned);
            }
        }

        // ---------- 段末：处理 A 类释放资源 ----------
        for (int i = 0; i < n; ++i) {
            if (p[i].need_release_A) {
                for (int res = 1; res <= m; ++res) {
                    if (p[i].owned & (1ULL << res)) {
                        if (owner[res] == i + 1) {
                            owner[res] = 0;
                        }
                        p[i].owned &= ~(1ULL << res);
                    }
                }
                p[i].need_release_A = false;
                p[i].fail_count = 0;
                p[i].first_apply_seg = -1;
            }
        }

        // ---------- 段末：任务运行与完成 ----------
        for (int i = 0; i < n; ++i) {
            if (p[i].state == 2) {
                p[i].rem--;
                if (p[i].rem == 0) {
                    if (p[i].j == p[i].k - 1) {
                        // 所有任务完成，释放资源并结束
                        for (int res = 1; res <= m; ++res) {
                            if (p[i].owned & (1ULL << res)) {
                                if (owner[res] == i + 1) {
                                    owner[res] = 0;
                                }
                                p[i].owned &= ~(1ULL << res);
                            }
                        }
                        p[i].state = 3;
                        p[i].end_seg = seg;
                    } else {
                        // 还有下一个任务，切换到资源申请状态
                        p[i].j++;
                        p[i].state = 1;
                        p[i].fail_count = 0;
                        p[i].first_apply_seg = -1;
                        p[i].rem = 0;
                        // 注意：之前占有的资源不释放
                    }
                }
            }
        }

        seg++;
    }

    // ---------- 输出结果 ----------
    for (int i = 0; i < n; ++i) {
        if (p[i].end_seg != -1) {
            cout << p[i].profit << " " << (p[i].end_seg - p[i].start_seg + 1) << "\n";
        } else {
            cout << p[i].profit << " -1\n";
        }
    }

    return 0;
}
```
