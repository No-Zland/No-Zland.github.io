最常用的基础形式，无命令行参数，仅执行内部逻辑：

```c
#include <stdio.h>

// 函数声明（被main调用，文档中函数示例扩展）
int sum(int a, int b);

int main(void) { // 标准无参数原型
    int a = 3, b = 5;
    int result = sum(a, b); // main调用sum函数（主调逻辑）
    
    printf("3 + 5 = %d\n", result); // 输出：8
    return 0; // 程序正常退出，返回0给操作系统
}

// 函数定义（被main调用的子函数）
int sum(int a, int b) {
    return a + b;
}
```
