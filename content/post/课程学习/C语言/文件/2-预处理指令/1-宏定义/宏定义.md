通过```#define```将标识符（宏名）与文本 / 表达式绑定，预处理阶段会将代码中所有宏名**替换为对应文本**，实现 “一处定义、多处复用”，分为 “无参数宏” 和 “带参数宏” 两类。

# 无参数宏：#define 宏名 替换文本

```c
#define 宏名 替换文本
```
  
替换文本 可省略，此时宏名仅作为 “标识”，如 ```#define``` DEBUG

示例：

```c
#include <stdio.h>// 无参数宏：定义常量（替换文本为数值/字符串）
#define PI 3.14159
#define MAX_SIZE 100
#define GREET "Hello C Preprocessor"

int main() {    

	// 预处理阶段：PI→3.14159，MAX_SIZE→100，GREET→"Hello C Preprocessor"    
	float radius = 5.0;    
	
	float area = PI * radius * radius; 
	   
	int arr[MAX_SIZE]; // 宏作为数组长度（编译前替换为100）
	        
	printf("圆面积：%.2f\n", area); // 输出：78.54    
	
	printf("%s\n", GREET);         // 输出：Hello C Preprocessor    
	
	return 0;
	
}
```


# 带参数宏：#define 宏名(参数列表) 替换表达式

```c
#define 宏名(参数列表) 替换表达式
```

参数列表无类型，表达式需加括号避免优先级问题

示例：带参数宏（处理简单逻辑）

```c
#include <stdio.h>
#define MAX(a, b) ((a) > (b) ? (a) : (b)) // 带参数宏：求两数最大值（表达式加括号避免优先级陷阱）
#define SQUARE(x) ((x) * (x)) // 带参数宏：计算平方（参数加括号，避免传入表达式时出错）

int main() {

    int x = 3, y = 5;    // 预处理替换：MAX(3,5)→((3)>(5)?(3):(5))，
    
    SQUARE( x + 2 ) → ( ( 3 + 2 ) * ( 3 + 2 ) )    
    
    printf("最大值：%d\n", MAX(x, y));    
    // 输出：5 
       
    printf("(x+2)的平方：%d\n", SQUARE( x + 2 ) ); 
    // 输出：25（若无括号会变成3+2*3+2=11）    
    
    return 0;
    
}
```

# 取消宏定义：#undef 宏名 

```c
#undef 宏名
```

取消已定义的宏，后续代码中宏名不再生效：

```c
#define NUM 100

printf("%d\n", NUM); // 输出：100

#undef NUM// printf("%d\n", NUM); // 错误：NUM未定义
```
