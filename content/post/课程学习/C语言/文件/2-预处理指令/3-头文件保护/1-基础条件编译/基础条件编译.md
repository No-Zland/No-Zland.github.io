通过条件判断控制 “哪些代码参与编译”，预处理阶段会删除不满足条件的代码，
实现 “跨平台适配、调试开关、代码版本控制”。
常用于嵌入式封装函数芯片型号选择、系统选择。

```c
#if 条件表达式
	
	满足条件的代码
	
#elif 条件表达式
	
	满足此条件的代码
	
#else
	
	所有条件不满足的代码
	
#endif
```

注意： 
**常量表达式**：条件表达式仅支持**常量表达式**（如宏、字面常量），如

```c
#if 1 // 正确
#if MAX_SIZE > 50 // 错误
```

**嵌套条件编译**：支持多层嵌套，需注意#endif 与#if/#ifdef 配对，避免遗漏：

```c
#define OS_WIN 1

#define DEBUG 1

#if OS_WIN 
	   
	#ifdef DEBUG    
	    
	    printf("Windows调试模式\n");    
		
	#else    
	    
	    printf("Windows发布模式\n");    
		
	#endif
	
#endif
```


**示例：** 跨平台适配

不同系统的头文件和函数不同（如 Windows 的 windows.h、Linux 的 unistd.h），用条件编译适配：

```c
#include <stdio.h>

// 判断当前系统宏（由编译器自动定义，如GCC定义__linux__，MSVC定义_WIN32）
#ifdef _WIN32   
 
	#include <windows.h> // Windows系统头文件   
 
	#define SLEEP(x) Sleep(x) // Windows休眠函数（毫秒）

#elif __linux__    

	#include <unistd.h> // Linux系统头文件
    
	#define SLEEP(x) sleep(x/1000) // Linux休眠函数（秒，转换为毫秒）

#else    

	#error "不支持当前操作系统" // 触发编译错误

#endif

int main() {

    printf("休眠3秒...\n");  
      
    SLEEP(3000); // 预处理阶段：Windows→Sleep(3000)，Linux→sleep(3)    
    
    printf("休眠结束\n");   
     
    return 0;
    
}
```
