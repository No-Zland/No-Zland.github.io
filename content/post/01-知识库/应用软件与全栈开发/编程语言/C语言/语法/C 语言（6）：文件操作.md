> C 语言文件操作通过标准库函数完成，核心头文件是 `<stdio.h>`。
> 
> 它用于和外部存储文件（文本文件、二进制文件）进行数据交互，实现数据持久化存储和外部数据读取。
> 
> 本文按“打开 → 读写 → 定位 → 关闭 → 错误处理”的主线展开，并给出文件复制、简单加密、批量处理等常见实战。

---

## 1. 文件操作流程与核心概念

**三步流程：**

1. **打开文件**：用 `fopen()` 建立程序与文件的连接，获取文件指针 `FILE *`。
2. **操作文件**：用读写函数（`fread()`、`fwrite()`、`fprintf()` 等）进行数据交互。
3. **关闭文件**：用 `fclose()` 释放资源，确保缓冲区数据同步到磁盘。

**关键概念：**

| 概念            | 说明                                               |
| ------------- | ------------------------------------------------ |
| 文件指针 `FILE *` | 系统维护的结构体指针，记录文件位置、打开模式、缓冲区状态等信息                  |
| 文本文件          | 以字符编码存储，可直接阅读，如 `.txt`、`.csv`                    |
| 二进制文件         | 以字节流存储，需按格式解析，如图片、视频、结构体数据                       |
| 缓冲区           | 系统为文件 I/O 分配的临时内存，减少磁盘读写次数；`fflush()` 可手动刷新输出缓冲区 |

## 2. 打开与关闭文件

### 2.1 `fopen()`：打开文件

```c
FILE *fopen(const char *filename, const char *mode);
```

- `filename`：文件路径，可以是相对路径（如 `test.txt`）或绝对路径。
- `mode`：打开模式，决定操作权限和文件类型。
- 返回值：成功返回文件指针，失败返回 `NULL`。必须检查返回值。

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("test.txt", "r");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }

    // 文件操作...

    fclose(fp);
    return 0;
}
```

### 2.2 `fclose()`：关闭文件

```c
int fclose(FILE *stream);
```

- 成功返回 `0`，失败返回 `EOF`。
- 关闭后文件指针不应再使用。
- 程序退出前应关闭所有打开的文件，否则缓冲区数据可能未写入磁盘。

### 2.3 文件打开模式

| 模式 | 类型 | 功能 | 文件不存在时 | 注意事项 |
|---|---|---|---|---|
| `"r"` | 文本 | 只读，指针在开头 | 打开失败 | 文件必须存在 |
| `"w"` | 文本 | 只写，清空或创建 | 创建新文件 | 会覆盖原内容，慎用 |
| `"a"` | 文本 | 追加写，指针在末尾 | 创建新文件 | 只能末尾追加 |
| `"r+"` | 文本 | 读写，指针在开头 | 打开失败 | 写入从当前位置覆盖 |
| `"w+"` | 文本 | 读写，清空或创建 | 创建新文件 | 适合重新写入并读取 |
| `"a+"` | 文本 | 读写，追加写 | 创建新文件 | 读之前需移动指针到开头 |
| `"rb"` | 二进制 | 只读 | 打开失败 | 用于图片、视频等 |
| `"wb"` | 二进制 | 只写，清空或创建 | 创建新文件 | 用于保存二进制数据 |
| `"ab"` | 二进制 | 追加写 | 创建新文件 | 适合追加二进制日志 |
| `"wx"` / `"xb"` | 文本/二进制 | 独占创建，文件已存在则失败 | 创建新文件 | C11 可选，避免覆盖已有文件 |

> 文本模式在部分平台会转换换行符，处理非文本文件时应使用二进制模式 `b`。

## 3. 文件读写方法

### 3.1 字符读写：`fgetc()` / `fputc()`

适合逐字符处理文本，例如统计字符数、逐字节加密。

```c
int fgetc(FILE *stream);          // 读取一个字符，返回 int，读到末尾或出错返回 EOF
int fputc(int c, FILE *stream);   // 写入一个字符
```

```c
#include <stdio.h>

int count_chars(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        perror("fopen");
        return -1;
    }

    int count = 0;
    int ch;
    while ((ch = fgetc(fp)) != EOF) {
        count++;
    }

    fclose(fp);
    return count;
}

int main(void) {
    printf("字符数：%d\n", count_chars("test.txt"));
    return 0;
}
```

### 3.2 格式化读写：`fprintf()` / `fscanf()`

用法类似 `printf` / `scanf`，多一个文件指针参数。

```c
int fprintf(FILE *stream, const char *format, ...);
int fscanf(FILE *stream, const char *format, ...);
```

```c
#include <stdio.h>

typedef struct {
    char name[20];
    int age;
    float score;
} Student;

int main(void) {
    Student stu = {"ZhangSan", 20, 95.5f};

    FILE *fp = fopen("students.txt", "w");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    fprintf(fp, "%s %d %.1f\n", stu.name, stu.age, stu.score);
    fclose(fp);

    fp = fopen("students.txt", "r");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    if (fscanf(fp, "%19s %d %f", stu.name, &stu.age, &stu.score) == 3) {
        printf("读取结果：%s %d %.1f\n", stu.name, stu.age, stu.score);
    }
    fclose(fp);

    return 0;
}
```

> `fscanf` 的 `%s` 遇空白停止，不能读取含空格的字符串。若字段含空格，建议用 `fgets` 读整行，再用 `sscanf` 解析，或改用明确分隔符的格式。

### 3.3 字符串读写：`fgets()` / `fputs()`

```c
char *fgets(char *str, int num, FILE *stream);
int fputs(const char *str, FILE *stream);
```

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("output.txt", "w");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    fputs("第一行文本\n", fp);
    fputs("第二行文本（含空格）\n", fp);
    fclose(fp);

    fp = fopen("output.txt", "r");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }

    char buf[1024];
    while (fgets(buf, sizeof(buf), fp) != NULL) {
        printf("%s", buf);  // fgets 保留换行符
    }
    fclose(fp);

    return 0;
}
```

- `fgets` 最多读取 `num - 1` 个字符，自动补 `'\0'`，遇换行或文件末尾停止。
- `fputs` 不自动添加换行符，需要手动写 `\n`。
- 不要使用 `gets()`，它没有长度限制，C11 已废弃。

### 3.4 二进制读写：`fread()` / `fwrite()`

直接操作字节流，适合图片、视频、结构体数组等。

```c
size_t fread(void *ptr, size_t size, size_t nmemb, FILE *stream);
size_t fwrite(const void *ptr, size_t size, size_t nmemb, FILE *stream);
```

返回值是**实际读写的块数**，不是字节数。

```c
#include <stdio.h>

typedef struct {
    int id;
    char name[20];
    float score;
} Student;

int main(void) {
    Student stus[3] = {
        {1, "Zhang San", 95.5f},
        {2, "Li Si", 88.0f},
        {3, "Wang Wu", 92.3f}
    };

    FILE *fp = fopen("students.bin", "wb");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    size_t written = fwrite(stus, sizeof(Student), 3, fp);
    printf("写入 %zu 个学生\n", written);
    fclose(fp);

    Student read_stus[3];
    fp = fopen("students.bin", "rb");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    size_t read_num = fread(read_stus, sizeof(Student), 3, fp);
    printf("读取 %zu 个学生\n", read_num);
    for (size_t i = 0; i < read_num; i++) {
        printf("%d %s %.1f\n", read_stus[i].id, read_stus[i].name, read_stus[i].score);
    }
    fclose(fp);

    return 0;
}
```

## 4. 文件指针定位

文件指针默认按顺序移动。需要随机读写时，可用以下函数调整位置。

| 函数 | 作用 |
|---|---|
| `fseek(fp, offset, whence)` | 移动文件指针 |
| `ftell(fp)` | 返回当前指针相对于文件开头的偏移量 |
| `rewind(fp)` | 重置指针到文件开头，等价于 `fseek(fp, 0, SEEK_SET)` |

`whence` 可取：

| 常量 | 含义 |
|---|---|
| `SEEK_SET` | 文件开头 |
| `SEEK_CUR` | 当前位置 |
| `SEEK_END` | 文件末尾 |

```c
#include <stdio.h>

long get_file_size(const char *filename) {
    FILE *fp = fopen(filename, "rb");
    if (fp == NULL) {
        perror("fopen");
        return -1;
    }

    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    fclose(fp);
    return size;
}

int main(void) {
    printf("文件大小：%ld 字节\n", get_file_size("test.txt"));
    return 0;
}
```

## 5. 常见实战

### 5.1 文件复制

用 `fread` / `fwrite` 分块复制，适合任意类型文件。缓冲区不要放在栈上过大，可用 `static` 或动态分配。

```c
#include <stdio.h>
#include <stdlib.h>

#define BUF_SIZE (1024 * 1024)  // 1MB

int copy_file(const char *src_path, const char *dst_path) {
    FILE *src = fopen(src_path, "rb");
    FILE *dst = fopen(dst_path, "wb");

    if (src == NULL || dst == NULL) {
        perror("open");
        if (src != NULL) fclose(src);
        if (dst != NULL) fclose(dst);
        return -1;
    }

    static char buf[BUF_SIZE];
    size_t read_num;
    while ((read_num = fread(buf, 1, BUF_SIZE, src)) > 0) {
        size_t write_num = fwrite(buf, 1, read_num, dst);
        if (write_num != read_num) {
            perror("write");
            fclose(src);
            fclose(dst);
            return -1;
        }
    }

    if (ferror(src)) {
        perror("read");
        fclose(src);
        fclose(dst);
        return -1;
    }

    fclose(src);
    fclose(dst);
    return 0;
}

int main(void) {
    if (copy_file("source.mp4", "copy.mp4") != 0) {
        return EXIT_FAILURE;
    }
    printf("复制成功\n");
    return 0;
}
```

### 5.2 简单异或加密 / 解密

异或同一个密钥，加密和解密使用同一函数。

```c
#include <stdio.h>

#define KEY 0x99

int xor_file(const char *in_path, const char *out_path) {
    FILE *in = fopen(in_path, "rb");
    FILE *out = fopen(out_path, "wb");
    if (in == NULL || out == NULL) {
        perror("open");
        if (in != NULL) fclose(in);
        if (out != NULL) fclose(out);
        return -1;
    }

    int ch;
    while ((ch = fgetc(in)) != EOF) {
        fputc(ch ^ KEY, out);
    }

    fclose(in);
    fclose(out);
    return 0;
}

int main(void) {
    xor_file("source.txt", "encrypted.txt");
    xor_file("encrypted.txt", "decrypted.txt");
    return 0;
}
```

> 这只是演示用简单异或，不具备实际安全强度。

### 5.3 批量处理文件（POSIX 示例）

`<dirent.h>` 是 POSIX 接口，Linux/macOS 可用；Windows 需使用其他 API。

```c
#include <stdio.h>
#include <dirent.h>
#include <string.h>

int count_chars(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        perror("fopen");
        return -1;
    }

    int count = 0;
    while (fgetc(fp) != EOF) {
        count++;
    }
    fclose(fp);
    return count;
}

int main(void) {
    DIR *dir = opendir(".");
    if (dir == NULL) {
        perror("opendir");
        return 1;
    }

    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        char *suffix = strrchr(entry->d_name, '.');
        if (suffix != NULL && strcmp(suffix, ".txt") == 0) {
            char path[256];
            snprintf(path, sizeof(path), "./%s", entry->d_name);
            int chars = count_chars(path);
            if (chars >= 0) {
                printf("%s: %d 字符\n", entry->d_name, chars);
            }
        }
    }

    closedir(dir);
    return 0;
}
```

## 6. 错误处理与排查

C 语言没有异常捕获机制，需要检查函数返回值，并配合错误处理函数。

| 函数 | 作用 |
|---|---|
| `perror(s)` | 打印系统错误信息，前缀为 `s` |
| `ferror(fp)` | 检查文件操作是否出错，非 0 表示出错 |
| `clearerr(fp)` | 清除错误标志和 EOF 标志 |

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("test.txt", "r");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }

    char buf[1024];
    if (fgets(buf, sizeof(buf), fp) == NULL) {
        if (ferror(fp)) {
            perror("fgets");
        } else {
            printf("文件为空或已到末尾\n");
        }
        fclose(fp);
        return 1;
    }

    fclose(fp);
    return 0;
}
```

### 常见错误与解决

| 现象 | 可能原因 | 解决 |
|---|---|---|
| `fopen` 返回 `NULL` | 路径错误、文件不存在、权限不足 | 检查路径、文件、权限 |
| 文本文件换行混乱 | 用文本模式打开二进制文件 | 非文本文件使用 `rb` / `wb` |
| 二进制文件损坏 | 用文本模式读写二进制 | 统一使用二进制模式 |
| 写入不生效 | 未关闭文件或未刷新缓冲区 | `fclose` 或 `fflush` |
| `fscanf` 读取不完整 | 格式与文件内容不匹配 | 确保格式符与数据一致，必要时用 `fgets` + `sscanf` |

## 7. 避坑清单

1. `fopen` 后必须检查返回值是否为 `NULL`。
2. 文件使用后必须 `fclose`，避免缓冲区数据丢失。
3. 处理图片、视频、结构体等非文本数据时使用二进制模式 `rb` / `wb`。
4. 大文件复制使用分块读写，缓冲区大小合理（如 1MB），避免一次性读入内存。
5. `fscanf` 的 `%s` 不能读取含空格字符串；需要时用 `fgets` 读整行再解析。
6. `fgets` 会保留换行符，必要时手动去除。
7. 不要使用 `gets()`；不要依赖 `fflush(stdin)`。
8. `fread` / `fwrite` 返回值是块数，不是字节数；应检查是否与预期一致。
9. 关闭文件后不要继续使用该 `FILE *` 指针。
10. 跨平台路径拼接推荐用 `snprintf`，不要手写死路径分隔符。
11. 加密、覆盖写等不可逆操作前先备份原文件。
12. 批量遍历目录的 `<dirent.h>` 是 POSIX 接口，Windows 下需换用其他 API。

---

## 8. 小结

文件操作的主线可以概括为：

- **打开**：`fopen` 获取 `FILE *`，检查 `NULL`。
- **读写**：字符用 `fgetc/fputc`，格式化用 `fprintf/fscanf`，字符串用 `fgets/fputs`，二进制用 `fread/fwrite`。
- **定位**：`fseek` / `ftell` / `rewind` 控制文件指针。
- **关闭**：`fclose` 释放资源并同步缓冲区。
- **错误处理**：检查返回值，配合 `perror`、`ferror`、`clearerr` 排查问题。

掌握这些函数后，文本日志、配置文件、二进制数据保存、文件复制等常见任务都可以用统一的三步流程完成。
