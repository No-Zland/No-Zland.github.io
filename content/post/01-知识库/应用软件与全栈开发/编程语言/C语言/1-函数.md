### 4-运算符优先级

- 实际编程中，建议通过 括号() 明确运算顺序，避免因优先级记忆偏差导致逻辑错误。
#####  说明

优先级越高，运算越先执行；优先级相同时，按 “结合性” 决定执行顺序（左结合：从左到右，右结合：从右到左）。

特殊场景：
- Python 无 ++/-- 运算符，用 i+=1/i-=1 替代；
- JavaScript 中 ===/!==（严格相等 / 不相等）优先级与 ==/!= 一致，仅避免隐式转换；
- 不同语言对部分运算符（如 ** 幂运算）的优先级可能有差异，要结合具体语言分析。

### 1-变量与常量
#### 1-变量
##### 0-变量
###### 变量
变量是程序中用于存储数据的内存空间，其核心要素包括定义（申请空间）、赋值（存储数据）和作用域（可访问范围）。

###### 1.1定义

定义格式：
```c
数据类型 变量名；// 可同时初始化，即定义时赋值
```

```c
int age;          // 定义int类型变量age（未初始化）

float score = 95.5;  // 定义float类型变量score，初始化值为95.5

char gender = 'M';   // 定义char类型变量gender，初始化值为'M'
```

###### 1.2 赋值

赋值是通过 = 运算符给已定义的变量赋予新值（区分 “初始化” 和 “赋值”：初始化是定义时赋值，赋值是定义后修改值）。

```c
int a;

a = 10;          // 给变量a赋值10

a = 20;          // 重新赋值为20（覆盖原有值）

float b = 3.14;

b = 6.28;        // 重新赋值为6.28
```

###### 1.3 作用域与生命周期

作用域指变量可被访问的代码范围，C 语言中主要有 3 类：

##### 1-局部变量
###### 局部变量
定义在函数内或代码块（{}）内，作用域仅限当前函数 / 块。

```c
void test() {
	
    int x = 5;  // 局部变量，仅在test()函数内有效
    
    if (1) {
	    
        int y = 10;  // 局部变量，仅在if块内有效
        
    }
    // printf("%d", y);  // 错误：y超出作用域
}
```

##### 2-全局变量
###### 全局变量
定义在所有函数外，作用域为整个程序（从定义处到文件结束）。

```c
int global = 100;  // 全局变量，所有函数均可访问

void func1() {
	
    printf("%d", global);  // 正确：可访问全局变量
    
}
```

##### 3-静态局部变量
###### 静态局部变量
```c
static 局部变量;
```

用 static 修饰的局部变量，作用域仍为当前函数 / 块，但生命周期为整个程序（值不会随函数调用结束而销毁）。

```c
void count() {
	
    static int cnt = 0;  // 静态局部变量，仅初始化一次
    
    cnt++;
    
    printf("%d ", cnt);  // 每次调用输出递增：1 2 3...
	
}
```

##### 1-类似容器结构
###### 类似容器类型
首先明确，C语言当中其实并没有内置容器类型，只是数组和字符串可以看作类似容器类型。
######  1.数组

```c
数据类型 数组名 [数组长度];
```

数组在内存中的存储方式，数组作为函数参数的使用（如传递数组地址）。

###### 1.1 一维数组：数据类型 数组名 [数组长度];

- 仅定义：int arr [5];（定义一个可存储 5 个 int 类型元素的一维数组）

- 完全初始化：int arr [5] = {1, 2, 3, 4, 5};

- 部分初始化：int arr [5] = {1, 2};（未初始化的元素自动为 0）

- 省略长度初始化：int arr [] = {1, 2, 3};（数组长度由初始化元素个数决定，此处为 3）。

###### 1.1.3 元素访问

通过下标访问，下标从 0 开始，格式为数组名 [下标]。

```c
arr[0] = 10; // 修改第1个元素为10

printf("%d", arr[2]); // 输出第3个元素
```

###### 1.1.4 输入输出示例


```c
#include <stdio.h>

int main() {
	
    int arr[5];
    
    // 输入
    printf("请输入5个整数：\n");
    
    for (int i = 0; i < 5; i++) {
	    
        scanf("%d", &arr[i]);
        
    }
    
    // 输出
    printf("数组元素：");
    
    for (int i = 0; i < 5; i++) {
	    
        printf("%d ", arr[i]);
        
    }
    
    return 0;
    
}
```

###### 1.2 二维数组：数据类型 数组名 [行数]  [列数];

```c
数据类型 数组名 [行数][列数];
```

又叫矩阵类数据存储。

```c
char matrix [2][3]; // （仅定义一个 2 行 3 列的二维数组，可存储 6 个 char 元素）

int matrix[2][3] = {{1, 2, 3}, {4, 5, 6}}; // 按行初始化

float matrix[2][3] = {1, 2, 3, 4, 5, 6};    // 按顺序初始化（效果与上相同）
```

###### 1.2.3 元素访问：数组名 [行下标]  [列下标]（）

通过行下标和列下标访问，格式为数组名 [行下标]  [列下标]（下标均从 0 开始）。

```c
matrix[0][1] = 10; // 修改第1行第2列元素为10

printf("%d", matrix[1][2]); // 输出第2行第3列元素
```

**示例：**

```c
#include <stdio.h>

int main() {
	
    int matrix[2][3];
    
    // 输入
    printf("请输入2行3列的整数：\n");
    
    for (int i = 0; i < 2; i++) {
    
        for (int j = 0; j < 3; j++) {
        
            scanf("%d", &matrix[i][j]);
            
        }
    }
    
    // 输出
    printf("二维数组元素：\n");
    
    for (int i = 0; i < 2; i++) {
	    
        for (int j = 0; j < 3; j++) {
        
            printf("%d ", matrix[i][j]);
        }
        
        printf("\n");
        
    }
    
    return 0;
    
}
```

###### 1.3 一维数组与二维数组的关系

二维数组可看作由多个一维数组组成：

对于 int matrix [2][3]，matrix [0] 是第一个一维数组（包含 matrix [0][0]、matrix [0][1]、matrix [0][2]），matrix [1] 是第二个一维数组。

可通过 “二维转一维”（按行 / 列将元素存储到一维数组）或 “一维转二维”（将一维数组元素按行列划分）实现两者转化。

**示例：**

```c
#include <stdio.h>

int main() {
	
    // 二维转一维
    int matrix[2][3] = {{1,2,3}, {4,5,6}};
    
    int arr[6];
    
    int index = 0;
    
    for (int i = 0; i < 2; i++) {
	    
	    for (int j = 0; j < 3; j++) {
            
            arr[index++] = matrix[i][j];
	        
        }
        
    }
    
    printf("二维转一维后：");
    
    for (int i = 0; i < 6; i++) {
	    
        printf("%d ", arr[i]);
	    
    }
    
    // 一维转二维
    int arr2[6] = {7,8,9,10,11,12};
    
    int matrix2[2][3];
    
    index = 0;
    
    for (int i = 0; i < 2; i++) {
    
        for (int j = 0; j < 3; j++) {
	        
            matrix2[i][j] = arr2[index++];
            
        }
        
    }
    
    printf("\n一维转二维后：\n");
    
    for (int i = 0; i < 2; i++) {
    
        for (int j = 0; j < 3; j++) {
        
            printf("%d ", matrix2[i][j]);
            
        }
        
        printf("\n");
        
    }
    
    return 0;
    
}
```

###### 2. 字符串

字符串本质是以 '\0'（ASCII 值为 0 的空字符）结尾的字符数组，'\0' 是字符串的结束标志。

示例：
```c
char str [] = "Hello"; // 等价于 char str [] = {'H', 'e', 'l', 'l', 'o', '\0'};
```

###### 2.1 常用字符串处理函数（需包含头文件 <string.h>）

###### 2.1.1 常用字符串处理函数

| 函数     | 原型                                          | 功能说明                                                                |
| ------ | ------------------------------------------- | ------------------------------------------------------------------- |
| strlen | size_t strlen(const char *s);               | 计算字符串 s 的长度（字符个数，不含 '\0'）                                           |
| strcpy | char *strcpy(char *dest, const char *src);  | 将字符串 src 复制到 dest 中（包含 '\0'），需保证 dest 有足够空间                         |
| strcmp | int strcmp(const char *s1, const char *s2); | 比较 s1 和 s2：相等返回 0；s1 大于 s2 返回正数；s1 小于 s2 返回负数                       |
| strcat | char *strcat(char *dest, const char *src);  | 将字符串 src 拼接至 dest 末尾（dest 原有 '\0' 被覆盖，最终只保留末尾的 '\0'），需保证 dest 有足够空间 |

###### 2.1.2 函数演示代码

```c
#include <stdio.h>
#include <string.h>

int main() {
	
    char str1[20] = "Hello";
    
    char str2[20] = "World";
    
    char str3[20], str4[20];
    
    // 1. strlen：获取字符串长度
    printf("str1长度：%zu\n", strlen(str1)); // 输出5
    
    printf("str2长度：%zu\n", strlen(str2)); // 输出5
    
    // 2. strcpy：复制字符串
    strcpy(str3, str1);
    
    printf("str3 = %s\n", str3); // 输出Hello
    
    // 3. strcmp：比较字符串
    int cmp = strcmp(str1, str2);
    
    if (cmp > 0)      printf("str1 > str2\n");
    
    else if (cmp == 0) printf("str1 == str2\n");
    
    else             printf("str1 < str2\n"); // 此处输出str1 < str2
    
    // 4. strcat：拼接字符串
    strcat(str1, " "); // 先给str1加个空格
    
    strcat(str1, str2);
    
    printf("str1拼接后：%s\n", str1); // 输出Hello World
    
    return 0;
	
}
```

#### 5-类型转换
##### 1-隐式转换（自动转换）
编译器自动进行的转换，遵循 “低精度→高精度” 原则（避免数据丢失），转换规则：

char/short → int → long → long long → float → double

```c
int a = 10;
float b = 3.14;
float c = a + b;  // a（int）隐式转为float，再与b相加，结果为float
```

注意：高精度→低精度的隐式转换可能导致数据丢失（如 double → int 会截断小数）。

##### 2-强制转换（显式转换）
用户通过 (目标类型) 显式指定的转换，格式：(目标类型) 表达式。

```c
double d = 5.99;
int e = (int)d;  // 强制转换：d（5.99）→ e（5，截断小数）
float f = 10.5f;
int g = (int)(f + 2.5);  // 先计算f+2.5=13.0，再强制转为int→13
```

注意：强制转换可能导致精度丢失或溢出，需谨慎使用。

### 2-输入与输出
#### 0-输入与输出
##### 定义

输入输出（I/O）是 C 语言与外部交互的核心，通过标准库 <stdio.h> 提供的函数实现；

按操作对象分为**标准 I/O**（键盘、屏幕）和**文件 I/O**（磁盘文件）。

##### 注意事项（通用避坑点）

1. 缓冲区影响：标准 I/O 默认使用缓冲区，输入输出不会立即生效，可通过 fflush(stdin)（清空输入缓冲区）、fflush(stdout)（刷新输出缓冲区）手动控制；

2. 格式符匹配：格式化 I/O 的格式符必须与数据类型严格对应，否则会出现乱码、内存异常（如 %d 对应 double）；

3. 地址传递：scanf()/fscanf() 读取普通变量时必须加 &（取地址），字符串数组无需加 &；

4. 换行符处理：%c、%[^\n] 对换行符敏感，需注意清空缓冲区残留的换行；

5. 文件 I/O 必关文件：使用 fopen() 打开的文件，必须用 fclose() 关闭，否则会导致资源泄漏、数据未同步到磁盘。

#### 1-输出
##### 字符
###### 1-输出单个字符到屏幕
###### 函数：putchar()

- **定义**：向标准输出设备（屏幕）输出一个字符，是简化版的 fputc(stdout)，无需指定文件指针，直接操作屏幕。

- **函数原型**：

```c
 int putchar(int c);
```

- 参数 c：待输出字符的 ASCII 码（可直接传入字符常量如 'A'，或 int 类型的 ASCII 值）；

- 返回值：成功返回输出的字符（ASCII 码），失败返回 EOF（-1）。

- **典型示例**：

```c
#include <stdio.h>

int main() {

    char ch1 = 'B';   
     
    int ch2 = 67; // 'C'的ASCII码    
    
    putchar('A');   // 输出字符'A'  
      
    putchar(ch1);   // 输出字符'B' 
       
    putchar(ch2);   // 输出字符'C'   
     
    putchar('\n');  // 输出换行符，换行  
      
    return 0;
    
}// 输出结果：ABC（换行）
```

- **特殊操作 / 注意事项**：

- 可直接输出转义字符，如 putchar('\t')（制表符）、putchar('\\')（反斜杠）；

- 虽然参数是 int 类型，但实际仅使用低 8 位（字符的 ASCII 码范围），传入超出范围的值会截断。

###### 2-输出单个字符到文件或屏幕
###### 函数：fputc()

- **定义**：向指定文件指针指向的设备（屏幕、文件等）输出一个字符，功能比 putchar() 灵活，支持文件输出。

- **函数原型**：
```c
int fputc(int c, FILE *stream);
```

- 参数 c：待输出字符的 ASCII 码；

- 参数 stream：文件指针（stdout 表示屏幕，或 fopen() 返回的文件指针）；

- 返回值：成功返回输出的字符，失败返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

int main() {   

	// 1. 输出到屏幕（等价于putchar()）    
	fputc('X', stdout); 
	   
	fputc('\n', stdout); 
	   
	// 2. 输出到文件    
	FILE *fp = fopen("output.txt", "w");    
	
	if (fp == NULL) {   
	 
	    perror("fopen error"); 
	           
	    return 1; 
	       
	}    
	
	fputc('Y', fp); // 向文件写入字符'Y'   
	 
	fclose(fp);   
	 
	return 0;
	
}// 屏幕输出：X（换行），文件output.txt中内容：Y
```

- **特殊操作 / 注意事项**：

- 可通过切换 stream 参数，同时输出到多个设备（如屏幕 + 文件）；

- 写入文件时，需确保文件以可写模式（w/a/r+ 等）打开，否则失败。

##### 字符串
###### 1-输出字符串到屏幕，自动换行
###### 函数：puts()

- **定义**：向标准输出设备（屏幕）输出字符串，自动在末尾添加换行符 \n，无需手动写入。

- **函数原型**：

int puts(const char *str);

- 参数 str：字符串指针（字符串数组名或字符串常量）；

- 返回值：成功返回非负值，失败返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

int main() {

    char str1[] = "Hello World";    
    
    puts(str1); // 输出字符串并换行    
    
    puts("C Language"); // 直接输出字符串常量并换行    
    
    puts(""); // 输出空字符串，仅换行    
    
    return 0;
    
}

// 输出结果：
// Hello World
// C Language
// （空行）
```

- **特殊操作 / 注意事项**：

- 自动添加换行符，若不需要换行，需用 printf("%s", str) 替代；

- 输出时遇到 \0 停止，若字符串无 \0 会输出垃圾值；

- 支持转义字符，如 puts("Hello\tWorld") 会输出带制表符的字符串。

###### 2-安全读取字符串，支持文件或键盘
###### 函数：char *fgets(char *str, int num, FILE *stream);

- **定义**：从指定设备（键盘、文件）读取字符串，可指定最大读取长度，避免缓冲区溢出，是安全的字符串读取函数，替代已废弃的 gets()。

- **函数原型**：
- 
```c
char *fgets(char *str, int num, FILE *stream);
```

- 参数 str：存储字符串的缓冲区（数组）；

- 参数 num：最大读取字符数（实际读取 num-1 个字符，留 1 位存 \0）；

- 参数 stream：文件指针（stdin 表示键盘，或 fopen() 返回的文件指针）；

- 返回值：成功返回 str 指针，失败或读到文件 / 输入末尾返回 NULL。

- **典型示例**：

```c
#include <stdio.h>

int main() {

    char buf[50]; 
       
    // 1. 从键盘读取（支持含空格）    
    printf("请输入内容（可含空格，最多49个字符）：");    
    
    fgets(buf, sizeof(buf), stdin); // sizeof(buf)获取缓冲区大小，避免硬编码    
    
    printf("你输入的：%s", buf); // fgets会保留输入时的回车符\n    
    
    // 2. 从文件读取    
    FILE *fp = fopen("str_input.txt", "r");    
    
    if (fp == NULL) {    
    
	    perror("fopen error");    
    
		return 1;    
    
    }    
    
    printf("从文件读取：");    
    
    while (fgets(buf, sizeof(buf), fp) != NULL) { // 循环读取所有行        
    
	    printf("%s", buf);    
	
	}    
	
	fclose(fp);    
	
	return 0;
	
}
```

- **特殊操作 / 注意事项**：

- 会保留输入 / 文件中的换行符 \n，若不需要可手动去除（如 buf[strcspn(buf, "\n")] = '\0'）；

- num 参数必须小于等于缓冲区大小，否则会导致缓冲区溢出；

- 读取键盘时，若输入字符数超过 num-1，会只读取前 num-1 个字符，剩余字符留在缓冲区；

- 读取文件时，遇换行符或文件末尾停止，每次调用读取一行。

##### 格式化（通用数据类型读写）
###### 1-格式化输出到屏幕
###### 函数：printf()

- **定义**：按自定义格式向标准输出设备（屏幕）输出数据，支持整数、浮点数、字符串等多种类型，是最常用的屏幕输出函数。

- **函数原型**：

```c
int printf(const char *format, ...);
```

- 参数 format：格式控制字符串，包含普通字符和格式符（如 %d/%f）；

- 参数 ...：可变参数列表，对应格式符的数据；

- 返回值：成功返回输出的字符总数，失败返回负数。

- **核心格式符**：

- 整数类型：%d（十进制）、%u（无符号十进制）、%x（十六进制小写）、%X（十六进制大写）、%o（八进制）；

- 浮点数类型：%f（默认保留 6 位小数）、%.nf（保留 n 位小数）、%e/%E（科学计数法）；

- 字符 / 字符串类型：%c（单个字符）、%s（字符串）；

- 特殊格式符：%p（指针地址）、%%（输出百分号）。

- **典型示例**：

```c
#include <stdio.h>

int main() {    

	int a = 100;   
	
	float b = 3.14159;    
	
	char c = 'M';    
	
	char str[] = "Hello C";    
	
	printf("整数：%d（十进制）、%x（十六进制）\n", a, a);    
	
	printf("浮点数：%.2f（保留2位）、%e（科学计数法）\n", b, b);    
	
	printf("字符：%c，字符串：%s\n", c, str);    
	
	printf("指针地址：%p\n", &a); // 输出变量a的内存地址    
	
	return 0;
	
}

// 输出结果：
// 整数：100（十进制）、64（十六进制）
// 浮点数：3.14（保留2位）、3.141590e+00（科学计数法）
// 字符：M，字符串：Hello C
// 指针地址：0x7ffee4b7c9ac（因系统而异）
```

- **特殊操作 / 注意事项**：

- 格式符与数据类型必须匹配，否则会出现乱码（如 %d 对应 float 类型）；

- 可通过格式符控制对齐方式：%10d（右对齐，占 10 位）、%-10d（左对齐）；

- 输出字符串时，%s 会遇到 \0 停止，若字符串无 \0 会读取垃圾值。

###### 2-格式化输出到文件
###### 函数：fprintf()

- **定义**：按自定义格式向指定文件指针指向的文件输出数据，功能与 printf() 一致，仅多了文件指针参数，支持文件写入。

- **函数原型**：
```c
int fprintf(FILE *stream, const char *format, ...);
```

- 参数 stream：文件指针（fopen() 返回，需以可写模式打开）；

- 参数 format：格式控制字符串（同 printf()）；

- 参数 ...：可变参数列表（同 printf()）；

- 返回值：成功返回输出的字符总数，失败返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

typedef struct {

    char name[20];  
      
    int age;  
      
    float score;
    
} Student;

int main() {  
  
	Student stu = {"Zhang San", 20, 95.5};
	    
	FILE *fp = fopen("student.txt", "w"); 
	   
	if (fp == NULL) {    
	
	    perror("fopen error");   
	         
	    return 1;    
	    
	}    // 格式化写入文件，格式符与printf一致    
	
	fprintf(fp, "姓名：%s\n年龄：%d\n成绩：%.1f\n", stu.name, stu.age, stu.score);  
	  
	printf("写入成功！\n");    
	
	fclose(fp);  
	  
	return 0;
	
}

// 文件student.txt内容：
// 姓名：Zhang San// 年龄：20// 成绩：95.5
```

- **特殊操作 / 注意事项**：

- 文件必须以可写模式打开（w/a/r+ 等），否则写入失败；

- 写入后需用 fclose(fp) 关闭文件，确保缓冲区数据同步到磁盘；

- 支持同时输出到屏幕和文件：fprintf(stdout, "内容") 等价于 printf("内容")。

#### 2-输入
##### 字符
###### 1-从键盘读取单个字符
###### 函数：int getchar(void);

- **定义**：从标准输入设备（键盘）读取一个字符，包括回车符 \n，是简化版的 fgetc(stdin)。

- **函数原型**：```
  
```c
int getchar(void);
```

- 无参数；

- 返回值：成功返回读取字符的 ASCII 码（int 类型），失败或读到输入结束（Ctrl+Z/Ctrl+D）返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

int main() {

    printf("请输入一个字符：");   
     
    int ch = getchar(); // 必须用int接收，避免EOF（-1）与char混淆  
      
    printf("你输入的字符是：");  
      
    putchar(ch);    
    
    printf("\n其ASCII码为：%d\n", ch);   
     
    return 0;
    
}// 输入：a → 输出：你输入的字符是：a  其ASCII码为：97
```

- **特殊操作 / 注意事项**：

- 读取时会等待键盘输入，直到按下回车才返回，输入的字符会暂存于输入缓冲区；

- 若缓冲区已有残留字符（如之前的输入未读完），会直接读取缓冲区内容，无需等待用户输入；

- 读取回车符 \n 时，返回值为 10（\n 的 ASCII 码）。

###### 2-从文件或键盘读取单个字符
###### 函数：fgetc()

- **定义**：从指定文件指针指向的设备（键盘、文件等）读取一个字符，功能比 getchar() 灵活，支持文件读取。

- **函数原型**

```c
int fgetc(FILE *stream);
```

- 参数 stream：文件指针（stdin 表示键盘，或 fopen() 返回的文件指针）；

- 返回值：成功返回读取字符的 ASCII 码，失败或读到文件末尾返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

int main() {    

	// 1. 从键盘读取（等价于getchar()）    
	printf("请输入字符：");    
	
	int ch1 = fgetc(stdin); 
	   
	printf("键盘输入：%c\n", ch1);    
	
	// 2. 从文件读取    
	FILE *fp = fopen("input.txt", "r");    
	
	if (fp == NULL) {    
	
	    perror("fopen error");   
	         
	    return 1;  
	      
	}    
	
	int ch2 = fgetc(fp); // 读取文件第一个字符    
	
	printf("文件输入：%c\n", ch2);    
	
	fclose(fp);    
	
	return 0;
	
}// 若input.txt中第一个字符为'Z'，输入键盘字符'a' → 输出：键盘输入：a  文件输入：Z
```

- **特殊操作 / 注意事项**：

- 读取文件时，文件指针会自动后移，多次调用可逐字符读取整个文件；

- 读取键盘时，同样受输入缓冲区影响，需注意清空残留字符（用 getchar() 或 fflush(stdin)，后者非标准但常用）。

##### 字符串
###### 1-输出字符串到文件或屏幕，不自动换行
###### 函数：int fputs(const char *str, FILE *stream);

- **定义**：向指定文件指针指向的设备（屏幕、文件）输出字符串，**不自动添加换行符**，需手动写入 \n。

- **函数原型**：
```c
int fputs(const char *str, FILE *stream);
```

- 参数 str：字符串指针；

- 参数 stream：文件指针（stdout 表示屏幕，或 fopen() 返回的文件指针）；

- 返回值：成功返回非负值，失败返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

int main() {

    char str[] = "Hello File I/O";    
    
    // 1. 输出到屏幕（不自动换行）    
    fputs(str, stdout);    
    
    fputs(" → 无换行\n", stdout); // 手动加\n换行    
    
    // 2. 输出到文件（不自动换行）    
    FILE *fp = fopen("str_output.txt", "w");    
    
    if (fp == NULL) {    
	    
	    perror("fopen error");        
	    
	    return 1;    
	    
	}    
	
	fputs(str, fp);    
	
	fputs("\n第二行字符串", fp); // 手动加\n换行    
	
	fclose(fp);    
	
	return 0;
	
}

// 屏幕输出：Hello File I/O → 无换行（换行）
// 文件str_output.txt内容：
// Hello File I/O
// 第二行字符串
```

- **特殊操作 / 注意事项**：

- 与 puts() 的核心区别：不自动换行，需手动添加 \n；

- 写入文件时，字符串中的 \0 不会被写入，仅写入 \0 前的字符；

- 支持输出空字符串（fputs("", stream)），但无实际意义。

###### 2-已废弃，不推荐使用
###### 函数：gets()

- **定义**：从键盘读取字符串，无长度限制，因存在严重的缓冲区溢出漏洞，C11 标准已废弃。

- **函数原型**：
```c
char *gets(char *str); // 已从标准库移除，部分编译器仍支持但不推荐）；
```

- **典型示例**：（仅作了解，禁止实际使用）

```c
#include <stdio.h>

int main() {

    char buf[10]; 
       
    gets(buf); 
    
    // 输入超过10个字符会导致缓冲区溢出，程序崩溃或内存异常    
    return 0;
    
}
```

- **特殊操作 / 注意事项**：

- 替代方案：用 fgets(buf, sizeof(buf), stdin) 或 scanf("%[^\n]", buf)（需注意缓冲区）；

- 缓冲区溢出会导致程序崩溃、数据泄露等安全问题，严禁在实际开发中使用。

##### 格式化
###### 1-格式化读取从键盘
###### 函数：scanf()

- **定义**：按自定义格式从标准输入设备（键盘）读取数据，支持多种类型，需传递变量地址（字符串数组除外）。

- **函数原型**：
```c
int scanf(const char *format, ...);
```

- 参数 format：格式控制字符串，包含格式符和可选分隔符（如空格、逗号）；

- 参数 ...：变量地址列表（普通变量加 &，字符串数组直接写数组名）；

- 返回值：成功返回读取的有效数据个数，失败返回 EOF，读取不匹配返回小于预期的个数。

- **核心格式符（对应数据类型）**：

- 整数类型：%d（int）、%u（unsigned int）、%ld（long）、%lld（long long）；

- 浮点数类型：%f（float）、%lf（double，输入必须用 %lf，输出可用 %f）；

- 字符 / 字符串类型：%c（char）、%s（字符串，遇空格 / 回车停止）、%[^\n]（字符串，遇回车停止，保留空格）。

- **典型示例**：

```c
#include <stdio.h>

int main() {

    int num;   
     
    double score;    
    
    char gender;   
     
    char name[20];  
      
    char addr[50];  
      
    // 1. 读取整数、浮点数、字符  
      
    printf("请输入整数 浮点数 字符（空格分隔）：");    
    
    scanf("%d %lf %c", &num, &score, &gender); // 注意double用%lf，变量加&
        
    // 2. 读取无空格字符串（%s）    
    
    printf("请输入姓名（无空格）：");   
     
    scanf("%s", name); // 字符串数组无需加&    
    
    // 3. 读取含空格字符串（%[^\n]），需先清空缓冲区回车   
     
    getchar(); // 清除上一步scanf残留的回车符  
      
    printf("请输入地址（可含空格）：");   
     
    scanf("%[^\n]", addr);    // 输出读取结果  
      
    printf("\n读取结果：\n");    
    
    printf("整数：%d，浮点数：%.2lf，字符：%c\n", num, score, gender); 
       
    printf("姓名：%s，地址：%s\n", name, addr);   
     
    return 0;
}
```

- **特殊操作 / 注意事项**：

- 普通变量必须加 &（取地址），否则编译报错或内存异常；字符串数组名本身是地址，无需加 &；

- %c 会读取缓冲区的残留字符（如回车、空格），需在读取前用 getchar() 清空；

- %[^\n] 是读取含空格字符串的关键，但其会保留缓冲区的回车，后续若再用 scanf 需先清空；

- 输入数据时，可用空格、回车、制表符作为分隔符，与格式符中的分隔符无关。

###### 2-格式化读取从文件
###### 函数：fscanf()

- **定义**：按自定义格式从指定文件指针指向的文件读取数据，功能与 scanf() 一致，仅多了文件指针参数，支持文件读取。

- **函数原型**：

```c
int fscanf(FILE *stream, const char *format, ...);
```

- 参数 stream：文件指针（fopen() 返回，需以可读模式打开）；

- 参数 format：格式控制字符串（同 scanf()）；

- 参数 ...：变量地址列表（同 scanf()）；

- 返回值：成功返回读取的有效数据个数，失败或读到文件末尾返回 EOF。

- **典型示例**：

```c
#include <stdio.h>

typedef struct {

    char name[20]; 
       
    int age;    
    
    float score;
    
} Student;

int main() {

    Student stu;    
    
    FILE *fp = fopen("student.txt", "r");  
      
    if (fp == NULL) {    
    
        perror("fopen error");  
              
        return 1;    
        
    }    // 格式化读取文件，跳过“姓名：”等前缀，仅取数据    
    
    fscanf(fp, "姓名：%s\n年龄：%d\n成绩：%.1f", stu.name, &stu.age, &stu.score); 
    
    printf("从文件读取结果：\n");    
    
    printf("姓名：%s，年龄：%d，成绩：%.1f\n", stu.name, stu.age, stu.score);  
      
    fclose(fp);   
     
    return 0;
    
}

// 输出结果：从文件读取结果：姓名：Zhang San，年龄：20，成绩：95.5
```

- **特殊操作 / 注意事项**：

- 格式符需与文件中数据的格式完全匹配（包括前缀、换行符），否则读取失败或数据错误；

- 读取字符串时，若文件中字符串含空格，需用 %[^\n] 格式符，同时注意文件中的换行位置；

- 读取大文件时，可循环调用 fscanf() 逐行 / 逐段读取，直到返回 EOF。

### 3-结构类型
#### 1-分支结构
##### 4-分支结构
控制程序的执行顺序，是实现逻辑的核心，分支结构用于根据条件选择不同的执行路径，C 语言中主要有 if 语句和 switch 语句。

###### 1. if语句
###### 1.1 单分支：if(条件表达式) 语句块

当条件表达式为真（非 0）时，执行语句块；否则跳过。

```c
if (条件表达式) {
    语句块;
}
````

示例：判断一个数是否为正数

```c
#include <stdio.h>

int main() {

    int num = 10;
    
    if (num > 0) {
	    
        printf("%d是正数\n", num);
        
    }
    
    return 0;
    
}
```

###### 1.2 双分支：if (条件表达式) {语句块1;} else {语句块2;}

条件表达式为真执行语句块 1，否则执行语句块 2。

```c
if (条件表达式) {

    语句块1;
    
} else {
	
    语句块2;
    
}
```

示例：判断一个数是奇数还是偶数

```c
#include <stdio.h>

int main() {
	
    int num = 7;
    
    if (num % 2 == 0) {
	    
        printf("%d是偶数\n", num);
        
    } else {
	    
        printf("%d是奇数\n", num);
        
    }
    
    return 0;
    
}
```

###### 1.3  多分支：if (条件表达式1) {语句块1;} else if (条件表达式2){语句块2;}...else {语句块n;}

依次判断条件，执行第一个为真的语句块；若所有条件都不满足，执行 else 后的语句块 n。

```c
if (条件表达式1) {

    语句块1;
    
} else if (条件表达式2) {

    语句块2;
    
}
...
else {

    语句块n;
    
}
```

示例：根据分数判断等级（A:≥90，B:80-89，C:60-79，D:＜60）

```c
#include <stdio.h>
int main() {

    int score = 85;
    
    if (score >= 90) {
    
        printf("等级：A\n");
        
    } else if (score >= 80) {
    
        printf("等级：B\n");
        
    } else if (score >= 60) {
    
        printf("等级：C\n");
        
    } else {
    
        printf("等级：D\n");
        
    }
    return 0;
}
```

###### 2. 多条件匹配：switch (条件表达式)

根据表达式的值，匹配对应的 case 常量，执行其后的语句块；break （见循环结构）用于跳出 switch 结构，否则会”贯穿“；default 处理所有未匹配的情况（可选）。

```c
switch (表达式) {
	
    case 常量1:
	    
        语句块1;
        
        break;
        
    case 常量2:
	    
        语句块2;
        
        break;
    ...
    default:
	    
        语句块n;
        
}
```

示例：根据数字输出对应星期

```c
#include <stdio.h>

int main() {

    int day = 3;
    
    switch (day) {
    
        case 1:printf("星期一\n");break;
        
        case 2:printf("星期二\n");break;
        
        case 3:printf("星期三\n");break;
        
        case 4:printf("星期四\n");break;
        
        case 5:printf("星期五\n");break;
        
        case 6:printf("星期六\n");break;
        
        case 7:printf("星期日\n");break;
        
        default: printf("输入无效\n");
        
    }
    
    return 0;
}
```

#### 2-循环结构
##### 6-循环结构
###### 1. 固定次数循环：for (初始化表达式; 条件表达式; 更新表达式) {循环体;}

执行流程：①执行初始化表达式 → ②判断条件表达式（为真则执行循环体，否则退出） → ③执行更新表达式 → 重复步骤②-③。

```c
for (初始化表达式; 条件表达式; 更新表达式) {
    循环体;
}
```

示例：输出 1 到 10 的整数

```c
#include <stdio.h>

int main() {
	
    for (int i = 1; i <= 10; i++) {
	    
        printf("%d ", i);
        
    }
    
    return 0;
    
}
```

###### 2. 先判断后执行：while (条件表达式) {循环体;}

执行流程：先判断条件表达式，**为真**则执行循环体，重复此过程；若条件一开始为假，循环体一次都不执行。

```c
while (条件表达式) {
    循环体;
}
```

示例：

```c
#include <stdio.h>

int main() {

    int i = 1, sum = 0;
    
    while (i <= 100) {
    
        sum += i;
        
        i++;
        
    }
    
    printf("1到100的和为：%d\n", sum);
    
    return 0;
    
}
```

###### 3. 先执行后判断do-while：do {循环体;} while (条件表达式);

执行流程：先执行循环体，再判断条件表达式；若条件为真，继续循环；否则退出。循环体至少执行一次。

```c
do {
    循环体;
} while (条件表达式);
```

示例：输入一个正整数（确保输入有效）

```c
#include <stdio.h>

int main() {

    int num;
    
    do {
    
        printf("请输入一个正整数：");
        
        scanf("%d", &num);
        
    } while (num <= 0);
    
    printf("你输入的正整数是：%d\n", num);
    
    return 0;
}
```

###### 4. 循环控制符

###### 4.1 跳出当前所在的循环：break 语句

作用：立即跳出当前所在的循环或 switch 结构，执行后续代码。

示例：当找到目标值时跳出循环

```c
#include <stdio.h>

int main() {
	
    int arr[] = {2, 4, 6, 8, 10};
    
    int target = 6, i;
    
    for (i = 0; i < 5; i++) {
	    
        if (arr[i] == target) {
	        
            printf("找到目标值，索引为：%d\n", i);
            
            break;  // 找到后跳出for循环
            
        }
        
    }
    
    return 0;
    
}
```

###### 4.2 跳过本次循环：continue 语句

作用：跳过本次循环的剩余代码，直接进入下一次循环的条件判断。

示例：输出 1 到 10 中的奇数

```c
#include <stdio.h>

int main() {
	
    for (int i = 1; i <= 10; i++) {
	    
        if (i % 2 == 0) {
	        
            continue;  // 偶数时跳过本次循环
            
        }
        
        printf("%d ", i);
        
    }
    
    return 0;
    
}
```
