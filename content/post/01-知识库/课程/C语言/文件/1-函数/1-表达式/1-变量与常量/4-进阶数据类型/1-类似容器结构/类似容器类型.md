首先明确，C语言当中其实并没有内置容器类型，只是数组和字符串可以看作类似容器类型。
#  1.数组

```c
数据类型 数组名 [数组长度];
```

数组在内存中的存储方式，数组作为函数参数的使用（如传递数组地址）。

## 1.1 一维数组：数据类型 数组名 [数组长度];

- 仅定义：int arr [5];（定义一个可存储 5 个 int 类型元素的一维数组）

- 完全初始化：int arr [5] = {1, 2, 3, 4, 5};

- 部分初始化：int arr [5] = {1, 2};（未初始化的元素自动为 0）

- 省略长度初始化：int arr [] = {1, 2, 3};（数组长度由初始化元素个数决定，此处为 3）。

### 1.1.3 元素访问

通过下标访问，下标从 0 开始，格式为数组名 [下标]。

```c
arr[0] = 10; // 修改第1个元素为10

printf("%d", arr[2]); // 输出第3个元素
```

### 1.1.4 输入输出示例


```c
#include <stdio.h>

int main() {
	
    int arr[5];
    
    // 输入
    printf("请输入5个整数：\n");
    
    for (int i = 0; i < 5; i++) {
	    
        scanf("%d", &arr[i]);
        
    }
    
    // 输出
    printf("数组元素：");
    
    for (int i = 0; i < 5; i++) {
	    
        printf("%d ", arr[i]);
        
    }
    
    return 0;
    
}
```

## 1.2 二维数组：数据类型 数组名 [行数]  [列数];

```c
数据类型 数组名 [行数][列数];
```

又叫矩阵类数据存储。

```c
char matrix [2][3]; // （仅定义一个 2 行 3 列的二维数组，可存储 6 个 char 元素）

int matrix[2][3] = {{1, 2, 3}, {4, 5, 6}}; // 按行初始化

float matrix[2][3] = {1, 2, 3, 4, 5, 6};    // 按顺序初始化（效果与上相同）
```

### 1.2.3 元素访问：数组名 [行下标]  [列下标]（）

通过行下标和列下标访问，格式为数组名 [行下标]  [列下标]（下标均从 0 开始）。

```c
matrix[0][1] = 10; // 修改第1行第2列元素为10

printf("%d", matrix[1][2]); // 输出第2行第3列元素
```

**示例：**

```c
#include <stdio.h>

int main() {
	
    int matrix[2][3];
    
    // 输入
    printf("请输入2行3列的整数：\n");
    
    for (int i = 0; i < 2; i++) {
    
        for (int j = 0; j < 3; j++) {
        
            scanf("%d", &matrix[i][j]);
            
        }
    }
    
    // 输出
    printf("二维数组元素：\n");
    
    for (int i = 0; i < 2; i++) {
	    
        for (int j = 0; j < 3; j++) {
        
            printf("%d ", matrix[i][j]);
        }
        
        printf("\n");
        
    }
    
    return 0;
    
}
```

## 1.3 一维数组与二维数组的关系

二维数组可看作由多个一维数组组成：

对于 int matrix [2][3]，matrix [0] 是第一个一维数组（包含 matrix [0][0]、matrix [0][1]、matrix [0][2]），matrix [1] 是第二个一维数组。

可通过 “二维转一维”（按行 / 列将元素存储到一维数组）或 “一维转二维”（将一维数组元素按行列划分）实现两者转化。

**示例：**

```c
#include <stdio.h>

int main() {
	
    // 二维转一维
    int matrix[2][3] = {{1,2,3}, {4,5,6}};
    
    int arr[6];
    
    int index = 0;
    
    for (int i = 0; i < 2; i++) {
	    
	    for (int j = 0; j < 3; j++) {
            
            arr[index++] = matrix[i][j];
	        
        }
        
    }
    
    printf("二维转一维后：");
    
    for (int i = 0; i < 6; i++) {
	    
        printf("%d ", arr[i]);
	    
    }
    
    // 一维转二维
    int arr2[6] = {7,8,9,10,11,12};
    
    int matrix2[2][3];
    
    index = 0;
    
    for (int i = 0; i < 2; i++) {
    
        for (int j = 0; j < 3; j++) {
	        
            matrix2[i][j] = arr2[index++];
            
        }
        
    }
    
    printf("\n一维转二维后：\n");
    
    for (int i = 0; i < 2; i++) {
    
        for (int j = 0; j < 3; j++) {
        
            printf("%d ", matrix2[i][j]);
            
        }
        
        printf("\n");
        
    }
    
    return 0;
    
}
```

# 2. 字符串

字符串本质是以 '\0'（ASCII 值为 0 的空字符）结尾的字符数组，'\0' 是字符串的结束标志。

示例：
```c
char str [] = "Hello"; // 等价于 char str [] = {'H', 'e', 'l', 'l', 'o', '\0'};
```

### 2.1 常用字符串处理函数（需包含头文件 <string.h>）

#### 2.1.1 常用字符串处理函数

| 函数     | 原型                                          | 功能说明                                                                |
| ------ | ------------------------------------------- | ------------------------------------------------------------------- |
| strlen | size_t strlen(const char *s);               | 计算字符串 s 的长度（字符个数，不含 '\0'）                                           |
| strcpy | char *strcpy(char *dest, const char *src);  | 将字符串 src 复制到 dest 中（包含 '\0'），需保证 dest 有足够空间                         |
| strcmp | int strcmp(const char *s1, const char *s2); | 比较 s1 和 s2：相等返回 0；s1 大于 s2 返回正数；s1 小于 s2 返回负数                       |
| strcat | char *strcat(char *dest, const char *src);  | 将字符串 src 拼接至 dest 末尾（dest 原有 '\0' 被覆盖，最终只保留末尾的 '\0'），需保证 dest 有足够空间 |

#### 2.1.2 函数演示代码

```c
#include <stdio.h>
#include <string.h>

int main() {
	
    char str1[20] = "Hello";
    
    char str2[20] = "World";
    
    char str3[20], str4[20];
    
    // 1. strlen：获取字符串长度
    printf("str1长度：%zu\n", strlen(str1)); // 输出5
    
    printf("str2长度：%zu\n", strlen(str2)); // 输出5
    
    // 2. strcpy：复制字符串
    strcpy(str3, str1);
    
    printf("str3 = %s\n", str3); // 输出Hello
    
    // 3. strcmp：比较字符串
    int cmp = strcmp(str1, str2);
    
    if (cmp > 0)      printf("str1 > str2\n");
    
    else if (cmp == 0) printf("str1 == str2\n");
    
    else             printf("str1 < str2\n"); // 此处输出str1 < str2
    
    // 4. strcat：拼接字符串
    strcat(str1, " "); // 先给str1加个空格
    
    strcat(str1, str2);
    
    printf("str1拼接后：%s\n", str1); // 输出Hello World
    
    return 0;
	
}
```
